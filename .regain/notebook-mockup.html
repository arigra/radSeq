1 · Real data

Read one RADIal frame and turn it into a dB map.

src/radial.py
import os
import numpy as np

DEFAULT_ROOT = "data/radial"


def power_map(sample, root=DEFAULT_ROOT):
    """RD power in dB, summed over receive channels."""
    x = np.load(os.path.join(root, "radar_FFT", f"fft_{sample:06d}.npy"))
    return 10 * np.log10((np.abs(x) ** 2).sum(axis=-1) + 1e-12)

2 · Training

Add noise to a map and teach the model to predict it back.

src/train.py
import torch
from src.diffusion import add_noise, v_target


def train_step(model, opt, x):
    t = torch.randint(0, 1000, (x.shape[0],))
    noise = torch.randn_like(x)
    x_t = add_noise(x, noise, t)
    # the model predicts v, not the noise itself
    loss = ((model(x_t, t) - v_target(x, noise, t)) ** 2).mean()
    opt.zero_grad()
    loss.backward()
    opt.step()
    return loss.item()


def train(model, loader, steps, lr=1e-4):
    opt = torch.optim.AdamW(model.parameters(), lr=lr)
    for step, batch in zip(range(steps), loader):
        loss = train_step(model, opt, batch["x"])
        if step % 100 == 0:
            print(step, loss)
